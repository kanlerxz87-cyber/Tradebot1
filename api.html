Request signing, as implemented

message = timestamp_ms + "GET" + path_without_query
signature = base64( ed25519_sign(private_key, message) ) // or RSA-PSS-SHA256
headers = KALSHI-ACCESS-KEY, KALSHI-ACCESS-SIGNATURE, KALSHI-ACCESS-TIMESTAMP
tolerance = timestamp within 30 seconds of Kalshi's clock
